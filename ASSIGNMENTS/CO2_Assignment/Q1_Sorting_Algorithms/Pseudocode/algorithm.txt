ALGORITHM SelectionSort(A, n)
  comparisons ← 0; swaps ← 0
  FOR i ← 0 TO n-2
    minIndex ← i
    FOR j ← i+1 TO n-1
      comparisons ← comparisons + 1
      IF A[j] < A[minIndex] THEN minIndex ← j
    END FOR
    IF minIndex ≠ i
      SWAP A[i], A[minIndex]
      swaps ← swaps + 1
    END IF
  END FOR
  RETURN A, comparisons, swaps

ALGORITHM BubbleSort(A, n)
  comparisons ← 0; swaps ← 0
  FOR i ← 0 TO n-2
    changed ← false
    FOR j ← 0 TO n-i-2
      comparisons ← comparisons + 1
      IF A[j] > A[j+1]
        SWAP A[j], A[j+1]; swaps ← swaps + 1
        changed ← true
      END IF
    END FOR
    IF changed = false THEN BREAK
  END FOR
  RETURN A, comparisons, swaps

ALGORITHM InsertionSort(A, n)
  comparisons ← 0; movements ← 0
  FOR i ← 1 TO n-1
    key ← A[i]; j ← i-1
    WHILE j ≥ 0
      comparisons ← comparisons + 1
      IF A[j] > key
        A[j+1] ← A[j]; movements ← movements + 1
        j ← j-1
      ELSE BREAK
    END WHILE
    A[j+1] ← key; movements ← movements + 1
  END FOR
  RETURN A, comparisons, movements
